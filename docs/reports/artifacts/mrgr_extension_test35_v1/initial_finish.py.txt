METRICS=('volume.relative_l1','volume.relative_l2','morphology.absolute_peak_ratio_error','morphology.peak_ratio','morphology.gt_peak_location_ratio','morphology.centroid_distance_mm','morphology.dice_0.1','morphology.dice_0.5','measurement.observed.relative_l1','measurement.observed.relative_l2','measurement.heldout.relative_l1','measurement.heldout.relative_l2')

def value(r,k):
    for p in k.split('.'):r=r[p]
    return r

def all_methods():return methods()+verified(BASE/'reuse_methods.json')
def root_for(m):return Path(m.get('prediction_root',BASE/'evaluation/test'/m['key']))

def independent(mm,lookup):
    source={r['case_id']:r for r in rows(DATA/'source_index.jsonl') if r['split']=='test'}
    count=0;errors={};computed={};allowed={}
    for m in mm:
        root=root_for(m)
        for r in rows(root/'manifest.jsonl'):allowed[str((root/r['prediction']).resolve())]='sealed_test_prediction'
    with TestGuard(gt_splits=['test']) as guard:
        guard.allowed.update(allowed)
        gt={c:np.load(DATA/r['gt_volume_64_path'],allow_pickle=False).astype(np.float64).squeeze() for c,r in source.items()}
        coords=np.linspace(0,.6,64)
        center=lambda c:np.array([np.dot(c.sum(axis=tuple(j for j in range(3) if j!=i)),coords)/c.sum() for i in range(3)])
        for m in mm:
            root=root_for(m)
            for r in rows(root/'manifest.jsonl'):
                path=root/r['prediction'];assert sha(path)==r['sha256'];g=gt[r['case_id']]
                with np.load(path,allow_pickle=False) as z:
                    assert set(z.files)=={'C0','C1'}
                    for stage in z.files:
                        p=z[stage].astype(np.float64);assert p.shape==(64,)*3 and np.isfinite(p).all() and p.min()>=0 and p.max()<=1
                        row=lookup[m['key'],stage,r['condition'],r['draw'],r['case_id']]
                        vals={'volume.relative_l1':float(np.abs(p-g).sum()/max(np.abs(g).sum(),1e-6)),
                            'volume.relative_l2':float(np.sqrt(np.square(p-g).sum()/max(np.square(g).sum(),1e-12))),
                            'morphology.peak_ratio':float(p.max()/g.max()),'morphology.gt_peak_location_ratio':float(p[np.unravel_index(g.argmax(),g.shape)]/g.max())}
                        if p.sum()>0:vals['morphology.centroid_distance_mm']=float(np.linalg.norm(center(p)-center(g))*1000)
                        for fraction in (.1,.5):
                            pm=p>=g.max()*fraction;gm=g>=g.max()*fraction
                            vals['morphology.dice_'+str(fraction)]=float(2*(pm&gm).sum()/(pm.sum()+gm.sum()))
                        for k,v in vals.items():
                            expected=value(row,k);assert math.isclose(v,expected,rel_tol=1e-8,abs_tol=1e-9),(m['key'],r['case_id'],k,v,expected)
                            errors[k]=max(errors.get(k,0.),abs(v-expected))
                        computed[m['key'],stage,r['condition'],r['draw'],r['case_id']]=vals['volume.relative_l1'];count+=1
            print('Independent field verification',m['key'],count,flush=True)
        assert not guard.denied and guard.counts['GT64_test']==35
    assert count==11*35*17*2
    freeze(BASE/'analysis/independent_fields.json',dict(at=now(),passed=True,methods=11,fields=count,cases=35,metrics_max_absolute_difference=errors,audit=guard.summary(),GT128_read=False))
    return computed

def panels_and_pairs(mm,lookup,computed):
    cases=verified(BASE/'registration.json')['cases']['test'];layouts=verified(BASE/'layouts.json')['test']
    panels=[(s['condition'],[s['draw']]) for s in layouts]+[(c,list(range(5))) for c in ('V3','V6','clustered-V4')]
    vectors={};summaries=[];csvrows=[]
    for m in mm:
        for stage in ('C0','C1'):
            for condition,draws in panels:
                stats={}
                for metric in METRICS:
                    array=[]
                    for case in cases:
                        vv=[value(lookup[m['key'],stage,condition,d,case],metric) for d in draws]
                        a=None if any(v is None for v in vv) else float(np.mean(vv));array.append(np.nan if a is None else a)
                        csvrows.append(dict(method=m['key'],arm=m['arm'],seed=m['seed'],stage=stage,condition=condition,draws=','.join(map(str,draws)),case_id=case,metric=metric,value=a))
                    x=np.array(array);finite=x[np.isfinite(x)];vectors[m['key'],stage,condition,tuple(draws),metric]=x
                    stats[metric]=dict(valid_cases=len(finite),mean=float(finite.mean()) if len(finite) else None,median=float(np.median(finite)) if len(finite) else None,p90=float(np.quantile(finite,.9)) if len(finite) else None,max=float(finite.max()) if len(finite) else None)
                summaries.append(dict(method=m['key'],arm=m['arm'],seed=m['seed'],stage=stage,condition=condition,draws=draws,metrics=stats))
    def key(arm,seed):return 'MRGR' if arm=='multiscale' and seed==20261011 else f'{arm}_seed{seed}'
    expressions=[]
    for label,terms in [('feedback-baseline',{'feedback_oari':1,'baseline':-1}),('feedback-zero_residual',{'feedback_oari':1,'feedback_oari_zero_residual':-1})]:
        expressions.append((label,20261011,[(key(a,20261011),c) for a,c in terms.items()]))
    for label,terms in [('B-A',{'uniform_fusion':1,'multiscale':-1}),('C-A',{'pruned_observability':1,'multiscale':-1}),('D-B',{'uniform_pruned_observability':1,'uniform_fusion':-1}),('D-C',{'uniform_pruned_observability':1,'pruned_observability':-1}),('(D-C)-(B-A)',{'uniform_pruned_observability':1,'pruned_observability':-1,'uniform_fusion':-1,'multiscale':1})]:
        for seeds in ([20261011],[20261019],[20261011,20261019]):
            expressions.append((label,seeds[0] if len(seeds)==1 else 'two_seed_case_mean',[(key(a,seed),c/len(seeds)) for seed in seeds for a,c in terms.items()]))
    indices=np.random.default_rng(20261020).integers(0,35,(20000,35));weights=np.stack([(indices==i).mean(1) for i in range(35)],1);paired=[];checks=0
    for condition,draws in [('Val-A-V4',[0]),('spread-V4',[0]),('V3',list(range(5))),('V6',list(range(5))),('clustered-V4',list(range(5)))]:
        for label,seed,terms in expressions:
            for metric in METRICS:
                delta=sum(c*vectors[k,'C1',condition,tuple(draws),metric] for k,c in terms);finite=np.isfinite(delta)
                r=dict(comparison=label,seed=seed,condition=condition,draws=draws,metric=metric,valid_pairs=int(finite.sum()),case_differences={case:float(v) if np.isfinite(v) else None for case,v in zip(cases,delta)})
                if finite.all():
                    ci=np.quantile(weights@delta,[.025,.975]).tolist()
                    r.update(delta=float(delta.mean()),ci95=ci,wins_lower=int((delta<-1e-8).sum()),ties=int((np.abs(delta)<=1e-8).sum()),losses_lower=int((delta>1e-8).sum()))
                    if metric=='volume.relative_l1':
                        independent_delta=np.array([sum(c*np.mean([computed[k,'C1',condition,d,case] for d in draws]) for k,c in terms) for case in cases])
                        np.testing.assert_allclose(delta,independent_delta,rtol=1e-10,atol=1e-10)
                        np.testing.assert_allclose(np.quantile(independent_delta[indices].mean(1),[.025,.975]),ci,rtol=1e-10,atol=1e-10);checks+=1
                else:r.update(delta=None,ci95=None,wins_lower=None,ties=None,losses_lower=None)
                paired.append(r)
    BASE.joinpath('analysis').mkdir(exist_ok=True)
    with (BASE/'analysis/per_case_metrics.csv').open('w',newline='') as f:
        w=csv.DictWriter(f,fieldnames=list(csvrows[0]),lineterminator='\n');w.writeheader();w.writerows(csvrows)
    jsonl(BASE/'analysis/paired_statistics.jsonl',paired)
    compact=[{k:v for k,v in r.items() if k!='case_differences'} for r in paired]
    write(BASE/'analysis/summary.json',dict(at=now(),split='supplementary_test35',cases=35,layouts=17,new_models=7,reused_models=4,panels=summaries,paired=compact,statistics=dict(seed=20261020,resamples=20000,unit='case; draw mean within case',unadjusted=True),limitations=['previously used35-case test; supplemental comparison, not an independent new holdout','feedback single seed','two-seed means descriptive; case bootstrap excludes training seed uncertainty','no test tuning or post-test selection','timings across GPUs/shared workloads not a matched speed comparison']))
    freeze(BASE/'analysis/independent_verification.json',dict(at=now(),passed=True,fields=11*35*17*2,models=11,cases=35,all_relative_L1_pairings_and_intervals_verified=checks,field_verification_sha256=sha(BASE/'analysis/independent_fields.json'),GT128_read=False))


def figures(mm):

    lookup={};allowed={}

        if m['key'] not in chosen:continue
        root=root_for(m)
        for r in rows(root/'manifest.jsonl'):

        guard.allowed.update(allowed)
        for index in (0,17,34):
            case=cases[index];g=np.load(DATA/source[case]['gt_volume_64_path'],allow_pickle=False).squeeze();y=np.unravel_index(g.argmax(),g.shape)[1]
            fig,axes=plt.subplots(7,2,figsize=(10,22),constrained_layout=True);norm=AsinhNorm(linear_width=float(g.max())*.02,vmin=0,vmax=float(g.max()))
